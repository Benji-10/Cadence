   152→export function EditSheet({
   153→  state,
   154→  open,

   158→  events,
   159→  defaultCalendarId,

   161→  const isMobile = useIsMobile();
   162→  const sheetSide: "bottom" | "right" = isMobile ? "bottom" : "right";
   163→
   164→  // Force a fresh form mount whenever the target event / create-defaults
   165→  // change so initial useState values are read from the latest props.
   166→  const formKey = useMemo(() => {
   167→    if (state?.mode === "edit" && state.event) return `edit-${state.event.id}`;
   168→    if (state?.mode === "create" && state.defaults) {
   169→      return `create-${state.defaults.start}-${state.defaults.end}`;
   170→    }
   171→    return "blank";
   172→  }, [state]);
   173→
   174→  return (
   175→    <Sheet open={open} onOpenChange={onOpenChange}>
   176→      <SheetContent
   177→        side={sheetSide}
   178→        className={cn(
   179→          "gap-0 p-0",
   180→          isMobile
   181→            ? "h-[92vh] max-h-[92vh] rounded-t-2xl"
   182→            : "w-full sm:max-w-md md:max-w-lg"
   183→        )}
   184→      >
   185→        <SheetHeader className="border-b border-border px-4 py-3">
   186→          <SheetTitle className="text-base">
   187→            {state?.mode === "edit" ? "Edit event" : "New event"}
   188→          </SheetTitle>
   189→          <SheetDescription className="sr-only">
   190→            {state?.mode === "edit"
   191→              ? "Update or delete this calendar event."
   192→              : "Create a new calendar event."}
   193→          </SheetDescription>
   194→        </SheetHeader>
   195→        <EditForm
   196→          key={formKey}
   197→          state={state}
   198→          onOpenChange={onOpenChange}
   199→          rangeStart={rangeStart}
   200→          rangeEnd={rangeEnd}
   201→          events={events}
   202→          defaultCalendarId={defaultCalendarId}
   203→        />
   204→      </SheetContent>

   206→  );

   217→
   218→function EditForm({
   219→  state,
   220→  onOpenChange,
   221→  rangeStart,
   222→  rangeEnd,
   223→  events,
   224→  defaultCalendarId,
   225→}: EditFormProps) {
   226→  const { data: calsData } = useCalendars();
   227→  const calendars: Calendar[] = calsData?.calendars ?? [];
   228→  const createMut = useCreateEvent();
   229→  const updateMut = useUpdateEvent();
   230→  const deleteMut = useDeleteEvent();
   231→  const suggestMut = useSuggest();
   232→
   233→  const tz = useMemo(
   234→    () => Intl.DateTimeFormat().resolvedOptions().timeZone || "UTC",
   235→    []
   236→  );
   237→  const settingsAlerts = useSettings((s) => s.defaultAlerts);
   238→  const addTemplate = useTemplates((s) => s.addTemplate);
   239→
   240→  // Initialise from `state` synchronously (lazy useState initializers).
   241→  const initial = useMemo(() => {
   242→    if (state?.mode === "edit" && state.event) {
   243→      const e = state.event;
   244→      return {
   245→        title: e.title,
   246→        allDay: e.allDay,
   247→        start: e.start,
   248→        end: e.end,
   249→        calendarId: e.calendarId,
   250→        color: e.color ?? null,
   251→        location: e.location ?? "",
   252→        travelMins: e.travelMins ?? 0,
   253→        alerts: e.alerts?.length ? e.alerts : settingsAlerts,
   254→        freq: e.recurrence ? e.recurrence.freq : "none",
   255→        interval: e.recurrence ? e.recurrence.interval : 1,
   256→        until: e.recurrence?.until ?? null,
   257→        priority: e.priority ?? 0,
   258→        timezone: e.timezone ?? tz,
   259→        notes: e.notes ?? "",
   260→        flexibility: e.flexibility,
   261→        minChunkMins: e.minChunkMins,
   262→        allowOverlap: e.allowOverlap,
   263→        locationType: e.locationType,
   264→      };
   265→    }
   266→    const d = state?.defaults;
   267→    const inferred = d?.start ? inferMetaFromTitle("") : inferMetaFromTitle("");
   268→    const startIso = d?.start ?? new Date().toISOString();
   269→    const endIso = d?.end ?? addMinutes(new Date(), 60).toISOString();
   270→    return {
   271→      title: "",
   272→      allDay: false,
   273→      start: startIso,
   274→      end: endIso,
   275→      calendarId: d?.calendarId ?? defaultCalendarId ?? calendars[0]?.id ?? "",
   276→      color: null,
   277→      location: "",
   278→      travelMins: 0,
   279→      alerts: settingsAlerts,
   280→      freq: "none",
   281→      interval: 1,
   282→      until: null,
   283→      priority: 0,
   284→      timezone: tz,
   285→      notes: "",
   286→      flexibility: inferred.flexibility,
   287→      minChunkMins: inferred.minChunkMins,
   288→      allowOverlap: inferred.allowOverlap,
   289→      locationType: inferred.locationType,
   290→    };
   291→  }, [state, tz, defaultCalendarId, calendars]);
   292→
   293→  const [title, setTitle] = useState(initial.title);
   294→  const [allDay, setAllDay] = useState(initial.allDay);
   295→  const [start, setStart] = useState(initial.start);
   296→  const [end, setEnd] = useState(initial.end);
   297→  const [calendarId, setCalendarId] = useState(initial.calendarId);
   298→  const [color, setColor] = useState<string | null>(initial.color);
   299→  const [location, setLocation] = useState(initial.location);
   300→  const [travelMins, setTravelMins] = useState(initial.travelMins);
   301→  const [alerts, setAlerts] = useState<number[]>(initial.alerts);
   302→  const [customAlert, setCustomAlert] = useState("");
   303→  const [freq, setFreq] = useState(initial.freq);
   304→  const [intervalN, setIntervalN] = useState(initial.interval);
   305→  const [until, setUntil] = useState<string | null>(initial.until);
   306→  const [priority, setPriority] = useState(initial.priority);
   307→  const [timezone, setTimezone] = useState(initial.timezone);
   308→  const [notes, setNotes] = useState(initial.notes);
   309→  const [flexibility, setFlexibility] = useState<Flexibility>(initial.flexibility);
   310→  const [minChunkMins, setMinChunkMins] = useState(initial.minChunkMins);
   311→  const [allowOverlap, setAllowOverlap] = useState(initial.allowOverlap);
   312→  const [locationType, setLocationType] = useState<LocationType>(initial.locationType);
   313→  const [advancedOpen, setAdvancedOpen] = useState(false);
   314→  const [locationDrawerOpen, setLocationDrawerOpen] = useState(false);
   315→  const [editingTitle, setEditingTitle] = useState(false);
   316→
   317→  const inferred = useMemo(() => inferMetaFromTitle(title), [title]);
   318→  const calendarsById = useMemo(
   319→    () => Object.fromEntries(calendars.map((c) => [c.id, c])),
   320→    [calendars]
   321→  );
   322→
   323→  const startDate = start ? parseISO(start) : new Date();
   324→  const endDate = end ? parseISO(end) : new Date();
   325→  const durationMins = differenceInMinutes(endDate, startDate);
   326→  const endBeforeStart = durationMins < 0;
   327→
   328→  const handleTitleChange = (v: string) => {
   329→    setTitle(v);
   330→    // In create mode, re-sync the inferred scheduling fields so the saved
   331→    // event picks up the right flexibility/min-chunk for its title.
   332→    if (state?.mode === "create" && v.trim()) {
   333→      const inf = inferMetaFromTitle(v);
   334→      setFlexibility(inf.flexibility);
   335→      setMinChunkMins(inf.minChunkMins);
   336→      setAllowOverlap(inf.allowOverlap);
   337→      setLocationType(inf.locationType);
   338→      if (!calendarId && inf.calendarKind) {
   339→        const match = calendars.find((c) => c.kind === inf.calendarKind);
   340→        if (match) setCalendarId(match.id);
   341→      }
   342→    }
   343→  };
   344→
   345→  const handleSave = async () => {
   346→    if (!title.trim()) {
   347→      toast.error("Give your event a title.");
   348→      return;
   349→    }
   350→    if (endBeforeStart && !allDay) {
   351→      toast.error("End time must be after start.");
   352→      return;
   353→    }
   354→    if (!calendarId) {
   355→      toast.error("Pick a calendar.");
   356→      return;
   357→    }
   358→    const recurrence: RecurrenceRule | null =
   359→      freq !== "none"
   360→        ? {
   361→            freq: freq as RecurrenceRule["freq"],
   362→            interval: Math.max(1, intervalN || 1),
   363→            until: until ?? undefined,
   364→          }
   365→        : null;
   366→    const payload: Partial<CalendarEvent> & {
   367→      title: string;
   368→      start: string;
   369→      end: string;
   370→      calendarId: string;
   371→    } = {
   372→      title: title.trim(),
   373→      start,
   374→      end: allDay ? start : end,
   375→      allDay,
   376→      calendarId,
   377→      location: location || null,
   378→      travelMins,
   379→      alerts,
   380→      recurrence,
   381→      priority,
   382→      timezone,
   383→      notes: notes || null,
   384→      flexibility,
   385→      minChunkMins,
   386→      allowOverlap,
   387→      locationType,
   388→      color: color ?? null,
   389→      category: inferred.category,
   390→    };
   391→    try {
   392→      if (state?.mode === "edit" && state.event) {
   393→        await updateMut.mutateAsync({ id: state.event.id, patch: payload });
   394→        toast.success("Event updated");
   395→      } else {
   396→        await createMut.mutateAsync(payload);
   397→        toast.success("Event added");
   398→      }
   399→      onOpenChange(false);
   400→    } catch (e) {
   401→      toast.error("Couldn't save", { description: String(e) });
   402→    }
   403→  };
   404→
   405→  const handleDelete = async () => {
   406→    if (!state?.event) return;
   407→    try {
   408→      await deleteMut.mutateAsync(state.event.id);
   409→      toast.success("Deleted");
   410→      onOpenChange(false);
   411→    } catch (e) {
   412→      toast.error("Couldn't delete", { description: String(e) });
   413→    }
   414→  };
   415→
   416→  // Duplicate the current event one day later (same time). Keeps all the
   417→  // intelligence metadata so the copy behaves the same way.
   418→  const handleDuplicate = async () => {
   419→    if (!state?.event) return;
   420→    const ev = state.event;
   421→    const startMs = parseISO(ev.start).getTime();
   422→    const endMs = parseISO(ev.end).getTime();
   423→    const dur = endMs - startMs;
   424→    const newStart = new Date(startMs + 24 * 60 * 60 * 1000);
   425→    const newEnd = new Date(newStart.getTime() + dur);
   426→    try {
   427→      const res = await createMut.mutateAsync({
   428→        title: ev.title,
   429→        start: newStart.toISOString(),
   430→        end: newEnd.toISOString(),
   431→        allDay: ev.allDay,
   432→        location: ev.location ?? undefined,
   433→        notes: ev.notes ?? undefined,
   434→        timezone: ev.timezone ?? undefined,
   435→        calendarId: ev.calendarId,
   436→        category: ev.category,
   437→        flexibility: ev.flexibility,
   438→        locationType: ev.locationType,
   439→        minChunkMins: ev.minChunkMins,
   440→        allowOverlap: ev.allowOverlap,
   441→        priority: ev.priority,
   442→        travelMins: ev.travelMins,
   443→        color: ev.color ?? undefined,
   444→        alerts: ev.alerts,
   445→        recurrence: ev.recurrence ?? undefined,
   446→      });
   447→      toast.success("Duplicated to " + format(newStart, "EEE d MMM, HH:mm"));
   448→      onOpenChange(false);
   449→      void res;
   450→    } catch (e) {
   451→      toast.error("Couldn't duplicate", { description: String(e) });
   452→    }
   453→  };
   454→
   455→  // Save the current event's title/duration/calendar/location as a reusable
   456→  // quick-add template.
   457→  const handleSaveAsTemplate = () => {
   458→    if (!title.trim()) {
   459→      toast.error("Type a title first.");
   460→      return;
   461→    }
   462→    const dur = Math.max(
   463→      15,
   464→      Math.round(
   465→        (parseISO(end).getTime() - parseISO(start).getTime()) / 60000
   466→      )
   467→    );
   468→    addTemplate({
   469→      title: title.trim(),
   470→      durationMins: dur,
   471→      calendarId,
   472→      location: location || undefined,
   473→      color,
   474→      category: inferred.category,
   475→      flexibility: inferred.flexibility,
   476→      locationType: inferred.locationType,
   477→    });
   478→    toast.success(`Saved "${title.trim()}" as a template.`);
   479→  };
   480→
   481→  // Apply a template's fields to the current form.
   482→  const applyTemplate = (t: {
   483→    title: string;
   484→    durationMins: number;
   485→    calendarId?: string;
   486→    location?: string;
   487→    color?: string | null;
   488→    category?: string;
   489→    flexibility?: string;
   490→    locationType?: string;
   491→  }) => {
   492→    setTitle(t.title);
   493→    if (t.location !== undefined) setLocation(t.location);
   494→    if (t.calendarId) setCalendarId(t.calendarId);
   495→    if (t.color !== undefined) setColor(t.color);
   496→    // Re-derive start/end: keep the current start, adjust end to the template duration.
   497→    const startMs = parseISO(start).getTime();
   498→    const newEnd = new Date(startMs + t.durationMins * 60000).toISOString();
   499→    setEnd(newEnd);
   500→    toast.success(`Applied "${t.title}" template.`);
   501→  };
   502→
   503→  const handleFindBestSlot = async () => {
   504→    if (!title.trim()) {
   505→      toast.error("Type a title first so I know what to schedule.");
   506→      return;
   507→    }
   508→    const dur = Math.max(15, Math.round(durationMins));
   509→    try {
   510→      const res = await suggestMut.mutateAsync({
   511→        task: { category: inferred.category, locationType, minChunkMins, flexibility },
   512→        durationMins: dur,
   513→        rangeStart,
   514→        rangeEnd,
   515→        aroundEvents: events,
   516→      });
   517→      if (!res.plan || res.plan.placements.length === 0) {
   518→        toast.error("No free slot found for this task.");
   519→        return;
   520→      }
   521→      const p = res.plan.placements[0];
   522→      setStart(p.start);
   523→      setEnd(p.end);
   524→      toast.success(`Placed: ${res.plan.reason}`);
   525→    } catch (e) {
   526→      toast.error("Couldn't find a slot", { description: String(e) });
   527→    }
   528→  };
   529→
   530→  const toggleAlert = (v: number) => {
   531→    setAlerts((cur) =>
   532→      cur.includes(v)
   533→        ? cur.filter((a) => a !== v)
   534→        : [...cur, v].sort((a, b) => a - b)
   535→    );
   536→  };
   537→  const addCustomAlert = () => {
   538→    const n = Number(customAlert);
   539→    if (!Number.isFinite(n) || n <= 0) {
   540→      toast.error("Enter a valid number of minutes.");
   541→      return;
   542→    }
   543→    // Store as negative (minutes-before start) to match the notifications engine.
   544→    toggleAlert(-n);
   545→    setCustomAlert("");
   546→  };
   547→
   548→  return (
   549→    <>
   550→      <div className="cal-scroll flex-1 overflow-y-auto px-4 py-4">
   551→        {/* Title — click-to-edit: shows as text until tapped */}
   552→        <div className="mb-3">
   553→          {editingTitle ? (
   554→            <Input
   555→              value={title}
   556→              onChange={(e) => handleTitleChange(e.target.value)}
   557→              onBlur={() => setEditingTitle(false)}
   558→              placeholder="Title"
   559→              autoFocus
   560→              className="h-12 border-0 px-0 text-lg font-semibold shadow-none focus-visible:ring-0"
   561→            />
   562→          ) : (
   563→            <button
   564→              onClick={() => setEditingTitle(true)}
   565→              className={cn(
   566→                "flex h-12 w-full items-center px-0 text-left text-lg font-semibold",
   567→                !title.trim() && "text-muted-foreground"
   568→              )}
   569→            >
   570→              {title.trim() || "Tap to add title…"}
   571→            </button>
   572→          )}
   573→          {/* Type-ahead: search previous events, select to copy all data */}
   574→          {state?.mode === "create" && title.trim().length >= 2 && (
   575→            <TitleAutocomplete
   576→              value={title}
   577→              onChange={() => {}}
   578→              onSelect={(ev) => {
   579→                setTitle(ev.title);
   580→                setAllDay(ev.allDay);
   581→                setCalendarId(ev.calendarId);
   582→                setColor(ev.color ?? null);
   583→                setLocation(ev.location ?? "");
